    45→function b64url(buf: ArrayBuffer): string {
    46→  return btoa(String.fromCharCode(...new Uint8Array(buf)))
    47→    .replace(/\+/g, "-")
    48→    .replace(/\//g, "_")
    49→    .replace(/=+$/, "");
    50→}
    51→
    52→function b64urlDecode(str: string): string {
    53→  return atob(str.replace(/-/g, "+").replace(/_/g, "/"));
    54→}

    56→// ── JWT (HS256) ──────────────────────────────────────────────
    57→
    58→async function createJWT(
    59→  payload: Record<string, unknown>,
    60→  secret: string
    61→): Promise<string> {
    62→  const header = b64url(
    63→    new TextEncoder().encode(JSON.stringify({ alg: "HS256", typ: "JWT" }))
    64→  );
    65→  const body = b64url(new TextEncoder().encode(JSON.stringify(payload)));
    66→  const msg = `${header}.${body}`;
    67→
    68→  const key = await crypto.subtle.importKey(
    69→    "raw",
    70→    new TextEncoder().encode(secret),
    71→    { name: "HMAC", hash: "SHA-256" },
    72→    false,
    73→    ["sign"]
    74→  );
    75→  const sig = b64url(
    76→    await crypto.subtle.sign("HMAC", key, new TextEncoder().encode(msg))
    77→  );
    78→
    79→  return `${msg}.${sig}`;
    80→}
    81→
    82→async function verifyJWT(
    83→  token: string,
    84→  secret: string
    85→): Promise<Record<string, unknown> | null> {
    86→  const parts = token.split(".");
    87→  if (parts.length !== 3) return null;
    88→
    89→  const [h, b, s] = parts;
    90→  const msg = `${h}.${b}`;
    91→
    92→  const key = await crypto.subtle.importKey(
    93→    "raw",
    94→    new TextEncoder().encode(secret),
    95→    { name: "HMAC", hash: "SHA-256" },
    96→    false,
    97→    ["verify"]
    98→  );
    99→
   100→  const sigStr = b64urlDecode(s);
   101→  const sigBuf = new Uint8Array(sigStr.length);
   102→  for (let i = 0; i < sigStr.length; i++) sigBuf[i] = sigStr.charCodeAt(i);
   103→
   104→  const valid = await crypto.subtle.verify(
   105→    "HMAC",
   106→    key,
   107→    sigBuf,
   108→    new TextEncoder().encode(msg)
   109→  );
   110→  if (!valid) return null;
   111→
   112→  const payload = JSON.parse(b64urlDecode(b));
   113→  if (payload.exp && payload.exp < Date.now() / 1000) return null;
   114→
   115→  return payload;
   116→}

   118→// ── AES-256-GCM + PBKDF2 (для ключ-файлов) ──────────────────
   119→// Используем 100_000 итераций (вместо 600_000) для совместимости
   120→// с лимитом CPU времени Cloudflare Worker (10мс на бесплатном тарифе).
   121→// 100K итераций — это всё равно ~3 года брутфорса на GPU.
   122→
   123→const PBKDF2_ITERATIONS = 100_000;
   124→const SALT_LENGTH = 16;
   125→const IV_LENGTH = 12;
   126→
   127→async function deriveKey(
   128→  password: string,
   129→  salt: Uint8Array
   130→): Promise<CryptoKey> {
   131→  const km = await crypto.subtle.importKey(
   132→    "raw",
   133→    new TextEncoder().encode(password),
   134→    "PBKDF2",
   135→    false,
   136→    ["deriveKey"]
   137→  );
   138→  return crypto.subtle.deriveKey(
   139→    {
   140→      name: "PBKDF2",
   141→      salt,
   142→      iterations: PBKDF2_ITERATIONS,
   143→      hash: "SHA-256",
   144→    },
   145→    km,
   146→    { name: "AES-GCM", length: 256 },
   147→    false,
   148→    ["encrypt", "decrypt"]
   149→  );
   150→}
   151→
   152→async function encrypt(
   153→  data: unknown,
   154→  password: string
   155→): Promise<string> {
   156→  const salt = crypto.getRandomValues(new Uint8Array(SALT_LENGTH));
   157→  const iv = crypto.getRandomValues(new Uint8Array(IV_LENGTH));
   158→  const key = await deriveKey(password, salt);
   159→  const ct = await crypto.subtle.encrypt(
   160→    { name: "AES-GCM", iv },
   161→    key,
   162→    new TextEncoder().encode(JSON.stringify(data))
   163→  );
   164→
   165→  const combined = new Uint8Array(salt.length + iv.length + ct.byteLength);
   166→  combined.set(salt, 0);
   167→  combined.set(iv, salt.length);
   168→  combined.set(new Uint8Array(ct), salt.length + iv.length);
   169→
   170→  return btoa(String.fromCharCode(...combined));
   171→}
   172→
   173→async function decrypt(base64: string, password: string): Promise<unknown> {
   174→  const combined = Uint8Array.from(atob(base64), (c) => c.charCodeAt(0));
   175→  if (combined.length < SALT_LENGTH + IV_LENGTH + 1)
   176→    throw new Error("Повреждённый файл ключа");
   177→
   178→  const salt = combined.slice(0, SALT_LENGTH);
   179→  const iv = combined.slice(SALT_LENGTH, SALT_LENGTH + IV_LENGTH);
   180→  const ct = combined.slice(SALT_LENGTH + IV_LENGTH);
   181→  const key = await deriveKey(password, salt);
   182→
   183→  let pt: ArrayBuffer;
   184→  try {
   185→    pt = await crypto.subtle.decrypt({ name: "AES-GCM", iv }, key, ct);
   186→  } catch {
   187→    throw new Error("Неверный пароль или повреждённый файл");
   188→  }
   189→
   190→  return JSON.parse(new TextDecoder().decode(pt));
   191→}

   193→// ── Anti-tampering token ─────────────────────────────────────
   194→
   195→async function generateToken(
   196→  role: string,
   197→  adminFolders: string[],
   198→  salt: string
   199→): Promise<string> {
   200→  const data = `${role}:${adminFolders.sort().join(",")}:${salt}`;
   201→  const buf = await crypto.subtle.digest(
   202→    "SHA-256",
   203→    new TextEncoder().encode(data)
   204→  );
   205→  return Array.from(new Uint8Array(buf))
   206→    .map((b) => b.toString(16).padStart(2, "0"))
   207→    .join("");
   208→}

   210→// ── Google Drive API ─────────────────────────────────────────
   211→
   212→async function driveFetch(
   213→  path: string,
   214→  params: Record<string, string>,
   215→  apiKey: string
   216→): Promise<Record<string, unknown>> {
   217→  const url = new URL(`https://www.googleapis.com/drive/v3${path}`);
   218→  url.searchParams.set("key", apiKey);
   219→  Object.entries(params).forEach(([k, v]) => url.searchParams.set(k, v));
   220→
   221→  const res = await fetch(url.toString());
   222→  if (!res.ok) {
   223→    const body = await res.json().catch(() => ({}));
   224→    throw new Error(
   225→      (body as Record<string, unknown>)?.error
   226→        ? String(((body as Record<string, unknown>).error as Record<string, unknown>)?.message || `Drive API error ${res.status}`)
   227→        : `Drive API error ${res.status}`
   228→    );
   229→  }
   230→  return res.json() as Promise<Record<string, unknown>>;
   231→}

   235→async function getAuth(
   236→  request: Request,
   237→  env: Env
   238→): Promise<Record<string, unknown> | null> {
   239→  const auth = request.headers.get("Authorization");
   240→  if (!auth?.startsWith("Bearer ")) return null;
   241→  return verifyJWT(auth.slice(7), env.JWT_SECRET);
   242→}

   246→async function handleAdminAuth(
   247→  request: Request,
   248→  env: Env
   249→): Promise<Response> {
   250→  const body = (await request.json()) as { password?: string };
   251→  const { password } = body;
   252→
   253→  if (!password || password !== env.ADMIN_PASSWORD) {
   254→    return json({ error: "Неверный пароль" }, 401);
   255→  }
   256→
   257→  const now = Math.floor(Date.now() / 1000);
   258→  const jwt = await createJWT(
   259→    {
   260→      role: "admin",
   261→      adminFolders: [],
   262→      iat: now,
   263→      exp: now + 7 * 24 * 3600, // 7 дней
   264→    },
   265→    env.JWT_SECRET
   266→  );
   267→
   268→  return json({ token: jwt, role: "admin" });
   269→}

   271→async function handleGuestAuth(
   272→  request: Request,
   273→  env: Env
   274→): Promise<Response> {
   275→  const body = (await request.json()) as {
   276→    keyFile?: string;
   277→    password?: string;
   278→  };
   279→  const { keyFile, password } = body;
   280→
   281→  if (!keyFile || !password) {
   282→    return json({ error: "Не указан ключ-файл или пароль" }, 400);
   283→  }
   284→
   285→  let kf: Record<string, unknown>;
   286→  try {
   287→    kf = (await decrypt(keyFile, password)) as Record<string, unknown>;
   288→  } catch (e) {
   289→    return json(
   290→      { error: e instanceof Error ? e.message : "Ошибка расшифровки" },
   291→      401
   292→    );
   293→  }
   294→
   295→  if (kf.v !== 4) {
   296→    return json({ error: "Неподдерживаемая версия ключ-файла" }, 400);
   297→  }
   298→
   299→  // Проверка анти-тамперинг токена (TOKEN_SALT скрыт на сервере!)
   300→  const expectedToken = await generateToken(
   301→    String(kf.role),
   302→    (kf.adminFolders as string[]) || [],
   303→    env.TOKEN_SALT
   304→  );
   305→  if (kf.token !== expectedToken) {
   306→    return json({ error: "Ключ-файл повреждён или был изменён" }, 401);
   307→  }
   308→
   309→  if (kf.role !== "guest") {
   310→    return json({ error: "Это не гостевой ключ-файл" }, 400);
   311→  }
   312→
   313→  const now = Math.floor(Date.now() / 1000);
   314→  const adminFolders = (kf.adminFolders as string[]) || [];
   315→  const jwt = await createJWT(
   316→    {
   317→      role: "guest",
   318→      adminFolders,
   319→      iat: now,
   320→      exp: now + 30 * 24 * 3600, // 30 дней
   321→    },
   322→    env.JWT_SECRET
   323→  );
   324→
   325→  return json({ token: jwt, role: "guest", adminFolders });
   326→}

   328→async function handleKeyFileGenerate(
   329→  request: Request,
   330→  env: Env
   331→): Promise<Response> {
   332→  const auth = await getAuth(request, env);
   333→  if (!auth || auth.role !== "admin") {
   334→    return json({ error: "Требуется авторизация администратора" }, 403);
   335→  }
   336→
   337→  const body = (await request.json()) as {
   338→    password?: string;
   339→    adminFolders?: string[];
   340→  };
   341→  const { password, adminFolders } = body;
   342→
   343→  if (!password || password.length < 4) {
   344→    return json(
   345→      { error: "Пароль должен быть не менее 4 символов" },
   346→      400
   347→    );
   348→  }
   349→
   350→  // Генерация анти-тамперинг токена (TOKEN_SALT скрыт на сервере!)
   351→  const token = await generateToken(
   352→    "guest",
   353→    adminFolders || [],
   354→    env.TOKEN_SALT
   355→  );
   356→
   357→  const keyFileData = {
   358→    v: 4,
   359→    role: "guest",
   360→    adminFolders: adminFolders || [],
   361→    token,
   362→  };
   363→
   364→  // Шифрование ключ-файла на сервере
   365→  const encrypted = await encrypt(keyFileData, password);
   366→
   367→  return json({ keyFile: encrypted });
   368→}

   370→async function handleFolders(
   371→  request: Request,
   372→  env: Env
   373→): Promise<Response> {
   374→  const auth = await getAuth(request, env);
   375→  if (!auth) {
   376→    return json({ error: "Требуется авторизация" }, 401);
   377→  }
   378→
   379→  // ── BFS: recursively discover ALL subfolders (not just top-level) ──
   380→  // The old code only fetched direct children of the root folder.
   381→  // This missed nested subfolders like "Видео / Как Шерон Стоун / Красотка в белом".
   382→  // Now we walk the entire folder tree level by level.
   383→  // SAFETY: stops after MAX_DEPTH levels, MAX_FOLDERS folders, or WALL_TIME_MS.
   384→  const allFolders: Record<string, unknown>[] = [];
   385→  let currentLevel = [env.DRIVE_ROOT_FOLDER_ID];
   386→  const visited = new Set<string>([env.DRIVE_ROOT_FOLDER_ID]);
   387→  const MAX_DEPTH = 10;      // Don't go deeper than 10 levels
   388→  const MAX_FOLDERS = 500;   // Don't discover more than 500 folders
   389→  const CHUNK_SIZE = 25;     // Max parent IDs per Drive API query (avoid URL length limits)
   390→  const WALL_TIME_MS = 20_000; // Stop BFS after 20s to avoid Cloudflare Worker timeout (30s)
   391→  const startTime = Date.now();
   392→
   393→  for (let depth = 0; depth < MAX_DEPTH && currentLevel.length > 0 && allFolders.length < MAX_FOLDERS; depth++) {
   394→    // Check wall-clock time before each level
   395→    if (Date.now() - startTime > WALL_TIME_MS) {
   396→      console.log(`[DMGA] BFS timeout after ${depth} levels, ${allFolders.length} folders, ${Date.now() - startTime}ms`);
   397→      break;
   398→    }
   399→
   400→    const nextLevel: string[] = [];
   401→
   402→    // Process current level in chunks to avoid Drive API query length limits
   403→    for (let i = 0; i < currentLevel.length; i += CHUNK_SIZE) {
   404→      // Check time before each chunk too
   405→      if (Date.now() - startTime > WALL_TIME_MS) break;
   406→
   407→      const chunk = currentLevel.slice(i, i + CHUNK_SIZE);
   408→      const parentQueries = chunk.map((id) => `'${id}' in parents`).join(" or ");
   409→      const query = `(${parentQueries}) and mimeType = 'application/vnd.google-apps.folder' and trashed = false`;
   410→
   411→      const data = await driveFetch(
   412→        "/files",
   413→        {
   414→          q: query,
   415→          fields: "files(id, name, createdTime, parents)",
   416→          pageSize: "500",
   417→          orderBy: "name",
   418→          supportsAllDrives: "true",
   419→          includeItemsFromAllDrives: "true",
   420→        },
   421→        env.DRIVE_API_KEY
   422→      );
   423→
   424→      const folders = (data.files as Record<string, unknown>[]) || [];
   425→      for (const folder of folders) {
   426→        const id = folder.id as string;
   427→        if (!visited.has(id)) {
   428→          visited.add(id);
   429→          allFolders.push(folder);
   430→          nextLevel.push(id);
   431→        }
   432→      }
   433→    }
   434→
   435→    currentLevel = nextLevel;
   436→  }
   437→
   438→  console.log(`[DMGA] BFS done: ${allFolders.length} folders in ${Date.now() - startTime}ms`);
   439→
   440→  let folders = allFolders;
   441→
   442→  // Фильтрация папок на сервере! Гость не видит admin-папки.
   443→  if (auth.role === "guest" && (auth.adminFolders as string[])?.length > 0) {
   444→    const adminFolders = auth.adminFolders as string[];
   445→    folders = folders.filter((f) => !adminFolders.includes(f.id as string));
   446→  }
   447→
   448→  return json({ folders });
   449→}

   451→async function handleFiles(
   452→  request: Request,
   453→  env: Env,
   454→  folderId: string
   455→): Promise<Response> {
   456→  const auth = await getAuth(request, env);
   457→  if (!auth) {
   458→    return json({ error: "Требуется авторизация" }, 401);
   459→  }
   460→
   461→  // Проверка доступа к папке на сервере!
   462→  if (
   463→    auth.role === "guest" &&
   464→    (auth.adminFolders as string[])?.includes(folderId)
   465→  ) {
   466→    return json({ error: "Доступ запрещён" }, 403);
   467→  }
   468→
   469→  const data = await driveFetch(
   470→    "/files",
   471→    {
   472→      q: `'${folderId}' in parents and (mimeType contains 'image/' or mimeType contains 'video/') and trashed = false`,
   473→      fields:
   474→        "files(id, name, mimeType, createdTime, modifiedTime, parents, size, webViewLink, webContentLink, thumbnailLink, imageMediaMetadata, videoMediaMetadata)",
   475→      pageSize: "200",
   476→      orderBy: "createdTime desc",
   477→      supportsAllDrives: "true",
   478→      includeItemsFromAllDrives: "true",
   479→    },
   480→    env.DRIVE_API_KEY
   481→  );
   482→
   483→  return json({ files: data.files || [] });
   484→}

   486→async function handleBatchFiles(
   487→  request: Request,
   488→  env: Env
   489→): Promise<Response> {
   490→  const auth = await getAuth(request, env);
   491→  if (!auth) {
   492→    return json({ error: "Требуется авторизация" }, 401);
   493→  }
   494→
   495→  // Parse folder IDs from query params: ?ids=id1,id2,id3
   496→  const url = new URL(request.url);
   497→  const idsParam = url.searchParams.get("ids");
   498→  if (!idsParam) {
   499→    return json({ error: "Не указаны ID папок (параметр ids)" }, 400);
   500→  }
   501→
   502→  const folderIds = idsParam.split(",").filter(Boolean);
   503→  if (folderIds.length === 0) {
   504→    return json({ error: "Пустой список ID папок" }, 400);
   505→  }
   506→
   507→  // Filter out admin-only folders for guest users
   508→  const adminFolders = (auth.adminFolders as string[]) || [];
   509→  const allowedFolderIds = auth.role === "guest" && adminFolders.length > 0
   510→    ? folderIds.filter((id) => !adminFolders.includes(id))
   511→    : folderIds;
   512→
   513→  if (allowedFolderIds.length === 0) {
   514→    return json({ files: {} });
   515→  }
   516→
   517→  // ── Chunked batch: split folder IDs into chunks to avoid Drive API query length limits ──
   518→  // With nested subfolders, the list of IDs can be very long (50+).
   519→  // Each `'ID' in parents` clause is ~35 chars, so ~25 IDs per query is safe.
   520→  const BATCH_CHUNK_SIZE = 25;
   521→  const filesByFolder: Record<string, Record<string, unknown>[]> = {};
   522→
   523→  for (let i = 0; i < allowedFolderIds.length; i += BATCH_CHUNK_SIZE) {
   524→    const chunk = allowedFolderIds.slice(i, i + BATCH_CHUNK_SIZE);
   525→    const parentQueries = chunk.map((id) => `'${id}' in parents`).join(" or ");
   526→    const query = `(${parentQueries}) and (mimeType contains 'image/' or mimeType contains 'video/') and trashed = false`;
   527→
   528→    const data = await driveFetch(
   529→      "/files",
   530→      {
   531→        q: query,
   532→        fields:
   533→          "files(id, name, mimeType, createdTime, modifiedTime, parents, size, webViewLink, webContentLink, thumbnailLink, imageMediaMetadata, videoMediaMetadata)",
   534→        pageSize: "1000",
   535→        orderBy: "createdTime desc",
   536→        supportsAllDrives: "true",
   537→        includeItemsFromAllDrives: "true",
   538→      },
   539→      env.DRIVE_API_KEY
   540→    );
   541→
   542→    // Group files by their parent folder
   543→    const rawFiles = (data.files as Record<string, unknown>[]) || [];
   544→    for (const file of rawFiles) {
   545→      const parents = (file.parents as string[]) || [];
   546→      for (const parentId of parents) {
   547→        if (allowedFolderIds.includes(parentId)) {
   548→          if (!filesByFolder[parentId]) filesByFolder[parentId] = [];
   549→          filesByFolder[parentId].push(file);
   550→        }
   551→      }
   552→    }
   553→  }
   554→
   555→  // Ensure all requested folders have an entry (even empty ones)
   556→  for (const folderId of allowedFolderIds) {
   557→    if (!filesByFolder[folderId]) {
   558→      filesByFolder[folderId] = [];
   559→    }
   560→  }
   561→
   562→  return json({ files: filesByFolder });
   563→}

   565→async function handleRootFiles(
   566→  request: Request,
   567→  env: Env
   568→): Promise<Response> {
   569→  const auth = await getAuth(request, env);
   570→  if (!auth) {
   571→    return json({ error: "Требуется авторизация" }, 401);
   572→  }
   573→
   574→  const data = await driveFetch(
   575→    "/files",
   576→    {
   577→      q: `'${env.DRIVE_ROOT_FOLDER_ID}' in parents and (mimeType contains 'image/' or mimeType contains 'video/') and trashed = false`,
   578→      fields:
   579→        "files(id, name, mimeType, createdTime, modifiedTime, parents, size, webViewLink, webContentLink, thumbnailLink, imageMediaMetadata, videoMediaMetadata)",
   580→      pageSize: "200",
   581→      orderBy: "createdTime desc",
   582→      supportsAllDrives: "true",
   583→      includeItemsFromAllDrives: "true",
   584→    },
   585→    env.DRIVE_API_KEY
   586→  );
   587→
   588→  return json({ files: data.files || [] });
   589→}

   591→// ── Media proxy — streams files from Google Drive ──────────
   592→
   593→async function handleMedia(
   594→  request: Request,
   595→  env: Env,
   596→  fileId: string
   597→): Promise<Response> {
   598→  // Auth: check token from query param or Authorization header
   599→  let token: string | null = null;
   600→  const url = new URL(request.url);
   601→  if (url.searchParams.has("token")) {
   602→    token = url.searchParams.get("token");
   603→  } else {
   604→    const auth = await getAuth(request, env);
   605→    if (!auth) {
   606→      return json({ error: "Требуется авторизация" }, 401);
   607→    }
   608→    token = request.headers.get("Authorization")?.slice(7) || null;
   609→  }
   610→
   611→  if (!token) {
   612→    return json({ error: "Требуется авторизация" }, 401);
   613→  }
   614→
   615→  // Verify JWT
   616→  const payload = await verifyJWT(token, env.JWT_SECRET);
   617→  if (!payload) {
   618→    return json({ error: "Недействительный токен" }, 401);
   619→  }
   620→
   621→  // ── HEAD request: probe if file is accessible ──
   622→  // The client sends a HEAD request to check if the video URL works
   623→  // before setting it as <video src>. This prevents the native
   624→  // browser error overlay from appearing.
   625→  if (request.method === "HEAD") {
   626→    const probeUrl = `https://www.googleapis.com/drive/v3/files/${fileId}?alt=media&key=${env.DRIVE_API_KEY}&supportsAllDrives=true&includeItemsFromAllDrives=true`;
   627→    try {
   628→      const probeRes = await fetch(probeUrl, {
   629→        headers: { "Range": "bytes=0-1" },
   630→      });
   631→      if (probeRes.ok || probeRes.status === 206) {
   632→        const contentType = probeRes.headers.get("Content-Type") || "application/octet-stream";
   633→        const contentLength = probeRes.headers.get("Content-Range")?.split("/")?.[1] || probeRes.headers.get("Content-Length") || "0";
   634→        return new Response(null, {
   635→          status: 200,
   636→          headers: {
   637→            ...CORS_HEADERS,
   638→            "Content-Type": contentType,
   639→            "Content-Length": contentLength,
   640→            "Accept-Ranges": "bytes",
   641→            "Cache-Control": "no-cache",
   642→            "X-DMGA-Status": "accessible",
   643→          },
   644→        });
   645→      } else {
   646→        let errorMsg = `Drive API error ${probeRes.status}`;
   647→        try {
   648→          const errBody = await probeRes.json();
   649→          errorMsg = (errBody as Record<string, unknown>)?.error
   650→            ? String(((errBody as Record<string, unknown>).error as Record<string, unknown>)?.message || errorMsg)
   651→            : errorMsg;
   652→        } catch {}
   653→        return json({ error: errorMsg }, probeRes.status as 400 | 401 | 403 | 404 | 500);
   654→      }
   655→    } catch (err) {
   656→      return json({ error: `Network error: ${err instanceof Error ? err.message : "unknown"}` }, 502);
   657→    }
   658→  }
   659→
   660→  // ── GET request: stream the file from Google Drive ──
   661→  // Forward Range header for video seeking support
   662→  const driveHeaders: Record<string, string> = {};
   663→  const rangeHeader = request.headers.get("Range");
   664→  if (rangeHeader) {
   665→    driveHeaders["Range"] = rangeHeader;
   666→  }
   667→
   668→  const driveUrl = `https://www.googleapis.com/drive/v3/files/${fileId}?alt=media&key=${env.DRIVE_API_KEY}&supportsAllDrives=true&includeItemsFromAllDrives=true`;
   669→
   670→  const driveRes = await fetch(driveUrl, { headers: driveHeaders });
   671→  if (!driveRes.ok) {
   672→    const body = await driveRes.text().catch(() => "");
   673→    return json(
   674→      { error: `Не удалось получить файл: ${driveRes.status}` },
   675→      driveRes.status as 400 | 401 | 403 | 404 | 500
   676→    );
   677→  }
   678→
   679→  // Stream response with CORS headers
   680→  const contentType = driveRes.headers.get("Content-Type") || "application/octet-stream";
   681→  const contentLength = driveRes.headers.get("Content-Length");
   682→  const contentRange = driveRes.headers.get("Content-Range");
   683→  const acceptRanges = driveRes.headers.get("Accept-Ranges");
   684→  const headers: Record<string, string> = {
   685→    ...CORS_HEADERS,
   686→    "Content-Type": contentType,
   687→    "Cache-Control": "public, max-age=3600", // 1h cache
   688→    "Accept-Ranges": acceptRanges || "bytes",
   689→  };
   690→  if (contentLength) headers["Content-Length"] = contentLength;
   691→  if (contentRange) headers["Content-Range"] = contentRange;
   692→
   693→  // 206 Partial Content for range requests
   694→  const isPartial = driveRes.status === 206;
   695→  return new Response(driveRes.body, {
   696→    status: isPartial ? 206 : 200,
   697→    headers,
   698→  });
   699→}

   701→// ── Thumbnail proxy — returns thumbnail image from Google Drive ──
   702→// Used as fallback when thumbnailLink is not available.
   703→// Supports ?token= query param for auth and ?sz= for size.
   704→
   705→async function handleThumbnail(
   706→  request: Request,
   707→  env: Env,
   708→  fileId: string
   709→): Promise<Response> {
   710→  // Auth: check token from query param or Authorization header
   711→  let token: string | null = null;
   712→  const url = new URL(request.url);
   713→  if (url.searchParams.has("token")) {
   714→    token = url.searchParams.get("token");
   715→  } else {
   716→    const auth = await getAuth(request, env);
   717→    if (!auth) {
   718→      return json({ error: "Требуется авторизация" }, 401);
   719→    }
   720→    token = request.headers.get("Authorization")?.slice(7) || null;
   721→  }
   722→
   723→  if (!token) {
   724→    return json({ error: "Требуется авторизация" }, 401);
   725→  }
   726→
   727→  // Verify JWT
   728→  const payload = await verifyJWT(token, env.JWT_SECRET);
   729→  if (!payload) {
   730→    return json({ error: "Недействительный токен" }, 401);
   731→  }
   732→
   733→  // Get thumbnail size from query param (default w400)
   734→  const sizeParam = url.searchParams.get("sz") || "w400";
   735→  const size = parseInt(sizeParam.replace("w", ""), 10) || 400;
   736→
   737→  // First, try to get the thumbnailLink from the Drive API
   738→  try {
   739→    const fileData = await driveFetch(
   740→      `/files/${fileId}`,
   741→      {
   742→        fields: "thumbnailLink,mimeType",
   743→        supportsAllDrives: "true",
   744→        includeItemsFromAllDrives: "true",
   745→      },
   746→      env.DRIVE_API_KEY
   747→    );
   748→
   749→    const thumbnailLink = fileData.thumbnailLink as string | undefined;
   750→    if (thumbnailLink) {
   751→      // Replace the size suffix in the thumbnail URL
   752→      const sizedUrl = thumbnailLink.replace(/=s\d+$/, `=s${size}`);
   753→      const thumbRes = await fetch(sizedUrl);
   754→      if (thumbRes.ok) {
   755→        const contentType = thumbRes.headers.get("Content-Type") || "image/jpeg";
   756→        const body = await thumbRes.arrayBuffer();
   757→        return new Response(body, {
   758→          status: 200,
   759→          headers: {
   760→            ...CORS_HEADERS,
   761→            "Content-Type": contentType,
   762→            "Cache-Control": "public, max-age=3600",
   763→            "Access-Control-Allow-Origin": "*",
   764→          },
   765→        });
   766→      }
   767→    }
   768→  } catch {
   769→    // thumbnailLink approach failed, try direct thumbnail
   770→  }
   771→
   772→  // Fallback: use drive.google.com/thumbnail
   773→  const thumbUrl = `https://drive.google.com/thumbnail?id=${fileId}&sz=w${size}`;
   774→  try {
   775→    const thumbRes = await fetch(thumbUrl, {
   776→      headers: { "Authorization": `Bearer ${env.DRIVE_API_KEY}` },
   777→    });
   778→    if (thumbRes.ok) {
   779→      const contentType = thumbRes.headers.get("Content-Type") || "image/jpeg";
   780→      const body = await thumbRes.arrayBuffer();
   781→      return new Response(body, {
   782→        status: 200,
   783→        headers: {
   784→          ...CORS_HEADERS,
   785→          "Content-Type": contentType,
   786→          "Cache-Control": "public, max-age=3600",
   787→        },
   788→      });
   789→    }
   790→  } catch {
   791→    // All thumbnail methods failed
   792→  }
   793→
   794→  return json({ error: "Не удалось получить миниатюру" }, 404);
   795→}

   797→// ── Main handler ─────────────────────────────────────────────
   798→
   799→export default {
   800→  async fetch(request: Request, env: Env): Promise<Response> {
   801→    // CORS preflight
   802→    if (request.method === "OPTIONS") {
   803→      return new Response(null, { headers: CORS_HEADERS });
   804→    }
   805→
   806→    const url = new URL(request.url);
   807→
   808→    try {
   809→      // Авторизация
   810→      if (url.pathname === "/auth/admin" && request.method === "POST") {
   811→        return handleAdminAuth(request, env);
   812→      }
   813→      if (url.pathname === "/auth/guest" && request.method === "POST") {
   814→        return handleGuestAuth(request, env);
   815→      }
   816→
   817→      // Генерация ключ-файлов (только для админа)
   818→      if (url.pathname === "/keyfile/generate" && request.method === "POST") {
   819→        return handleKeyFileGenerate(request, env);
   820→      }
   821→
   822→      // Данные — требуют авторизацию
   823→      if (url.pathname === "/folders" && request.method === "GET") {
   824→        return handleFolders(request, env);
   825→      }
   826→      if (url.pathname === "/files/root" && request.method === "GET") {
   827→        return handleRootFiles(request, env);
   828→      }
   829→      if (url.pathname === "/files/batch" && request.method === "GET") {
   830→        return handleBatchFiles(request, env);
   831→      }
   832→      if (url.pathname.startsWith("/files/") && request.method === "GET") {
   833→        const folderId = url.pathname.slice(7);
   834→        return handleFiles(request, env, folderId);
   835→      }
   836→
   837→      // Миниатюры — проксирует thumbnails из Google Drive
   838→      if (url.pathname.startsWith("/thumbnail/") && request.method === "GET") {
   839→        const fileId = url.pathname.slice(11);
   840→        return handleThumbnail(request, env, fileId);
   841→      }
   842→
   843→      // Медиа-прокси — стримит файлы Google Drive через Worker
   844→      // Используется для видео и полноразмерных изображений
   845→      // Поддерживает GET (стриминг) и HEAD (проверка доступности)
   846→      if (url.pathname.startsWith("/media/") && (request.method === "GET" || request.method === "HEAD")) {
   847→        const fileId = url.pathname.slice(7);
   848→        return handleMedia(request, env, fileId);
   849→      }
   850→
   851→      // Health check
   852→      if (url.pathname === "/") {
   853→        return json({ status: "ok", service: "DMGA API Proxy" });
   854→      }
   855→
   856→      return json({ error: "Not found" }, 404);
   857→    } catch (err: unknown) {
   858→      const message = err instanceof Error ? err.message : "Internal server error";
   859→      return json({ error: message }, 500);
   860→    }
   861→  },
   862→};
