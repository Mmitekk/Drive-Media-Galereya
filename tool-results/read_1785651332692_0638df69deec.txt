    45→    45→function b64url(buf: ArrayBuffer): string {
    46→    46→  return btoa(String.fromCharCode(...new Uint8Array(buf)))
    47→    47→    .replace(/\+/g, "-")
    48→    48→    .replace(/\//g, "_")
    49→    49→    .replace(/=+$/, "");
    50→    50→}
    51→    51→
    52→    52→function b64urlDecode(str: string): string {
    53→    53→  return atob(str.replace(/-/g, "+").replace(/_/g, "/"));
    54→    54→}

    56→    56→// ── JWT (HS256) ──────────────────────────────────────────────
    57→    57→
    58→    58→async function createJWT(
    59→    59→  payload: Record<string, unknown>,
    60→    60→  secret: string
    61→    61→): Promise<string> {
    62→    62→  const header = b64url(
    63→    63→    new TextEncoder().encode(JSON.stringify({ alg: "HS256", typ: "JWT" }))
    64→    64→  );
    65→    65→  const body = b64url(new TextEncoder().encode(JSON.stringify(payload)));
    66→    66→  const msg = `${header}.${body}`;
    67→    67→
    68→    68→  const key = await crypto.subtle.importKey(
    69→    69→    "raw",
    70→    70→    new TextEncoder().encode(secret),
    71→    71→    { name: "HMAC", hash: "SHA-256" },
    72→    72→    false,
    73→    73→    ["sign"]
    74→    74→  );
    75→    75→  const sig = b64url(
    76→    76→    await crypto.subtle.sign("HMAC", key, new TextEncoder().encode(msg))
    77→    77→  );
    78→    78→
    79→    79→  return `${msg}.${sig}`;
    80→    80→}
    81→    81→
    82→    82→async function verifyJWT(
    83→    83→  token: string,
    84→    84→  secret: string
    85→    85→): Promise<Record<string, unknown> | null> {
    86→    86→  const parts = token.split(".");
    87→    87→  if (parts.length !== 3) return null;
    88→    88→
    89→    89→  const [h, b, s] = parts;
    90→    90→  const msg = `${h}.${b}`;
    91→    91→
    92→    92→  const key = await crypto.subtle.importKey(
    93→    93→    "raw",
    94→    94→    new TextEncoder().encode(secret),
    95→    95→    { name: "HMAC", hash: "SHA-256" },
    96→    96→    false,
    97→    97→    ["verify"]
    98→    98→  );
    99→    99→
   100→   100→  const sigStr = b64urlDecode(s);
   101→   101→  const sigBuf = new Uint8Array(sigStr.length);
   102→   102→  for (let i = 0; i < sigStr.length; i++) sigBuf[i] = sigStr.charCodeAt(i);
   103→   103→
   104→   104→  const valid = await crypto.subtle.verify(
   105→   105→    "HMAC",
   106→   106→    key,
   107→   107→    sigBuf,
   108→   108→    new TextEncoder().encode(msg)
   109→   109→  );
   110→   110→  if (!valid) return null;
   111→   111→
   112→   112→  const payload = JSON.parse(b64urlDecode(b));
   113→   113→  if (payload.exp && payload.exp < Date.now() / 1000) return null;
   114→   114→
   115→   115→  return payload;
   116→   116→}

   118→   118→// ── AES-256-GCM + PBKDF2 (для ключ-файлов) ──────────────────
   119→   119→// Используем 100_000 итераций (вместо 600_000) для совместимости
   120→   120→// с лимитом CPU времени Cloudflare Worker (10мс на бесплатном тарифе).
   121→   121→// 100K итераций — это всё равно ~3 года брутфорса на GPU.
   122→   122→
   123→   123→const PBKDF2_ITERATIONS = 100_000;
   124→   124→const SALT_LENGTH = 16;
   125→   125→const IV_LENGTH = 12;
   126→   126→
   127→   127→async function deriveKey(
   128→   128→  password: string,
   129→   129→  salt: Uint8Array
   130→   130→): Promise<CryptoKey> {
   131→   131→  const km = await crypto.subtle.importKey(
   132→   132→    "raw",
   133→   133→    new TextEncoder().encode(password),
   134→   134→    "PBKDF2",
   135→   135→    false,
   136→   136→    ["deriveKey"]
   137→   137→  );
   138→   138→  return crypto.subtle.deriveKey(
   139→   139→    {
   140→   140→      name: "PBKDF2",
   141→   141→      salt,
   142→   142→      iterations: PBKDF2_ITERATIONS,
   143→   143→      hash: "SHA-256",
   144→   144→    },
   145→   145→    km,
   146→   146→    { name: "AES-GCM", length: 256 },
   147→   147→    false,
   148→   148→    ["encrypt", "decrypt"]
   149→   149→  );
   150→   150→}
   151→   151→
   152→   152→async function encrypt(
   153→   153→  data: unknown,
   154→   154→  password: string
   155→   155→): Promise<string> {
   156→   156→  const salt = crypto.getRandomValues(new Uint8Array(SALT_LENGTH));
   157→   157→  const iv = crypto.getRandomValues(new Uint8Array(IV_LENGTH));
   158→   158→  const key = await deriveKey(password, salt);
   159→   159→  const ct = await crypto.subtle.encrypt(
   160→   160→    { name: "AES-GCM", iv },
   161→   161→    key,
   162→   162→    new TextEncoder().encode(JSON.stringify(data))
   163→   163→  );
   164→   164→
   165→   165→  const combined = new Uint8Array(salt.length + iv.length + ct.byteLength);
   166→   166→  combined.set(salt, 0);
   167→   167→  combined.set(iv, salt.length);
   168→   168→  combined.set(new Uint8Array(ct), salt.length + iv.length);
   169→   169→
   170→   170→  return btoa(String.fromCharCode(...combined));
   171→   171→}
   172→   172→
   173→   173→async function decrypt(base64: string, password: string): Promise<unknown> {
   174→   174→  const combined = Uint8Array.from(atob(base64), (c) => c.charCodeAt(0));
   175→   175→  if (combined.length < SALT_LENGTH + IV_LENGTH + 1)
   176→   176→    throw new Error("Повреждённый файл ключа");
   177→   177→
   178→   178→  const salt = combined.slice(0, SALT_LENGTH);
   179→   179→  const iv = combined.slice(SALT_LENGTH, SALT_LENGTH + IV_LENGTH);
   180→   180→  const ct = combined.slice(SALT_LENGTH + IV_LENGTH);
   181→   181→  const key = await deriveKey(password, salt);
   182→   182→
   183→   183→  let pt: ArrayBuffer;
   184→   184→  try {
   185→   185→    pt = await crypto.subtle.decrypt({ name: "AES-GCM", iv }, key, ct);
   186→   186→  } catch {
   187→   187→    throw new Error("Неверный пароль или повреждённый файл");
   188→   188→  }
   189→   189→
   190→   190→  return JSON.parse(new TextDecoder().decode(pt));
   191→   191→}

   193→   193→// ── Anti-tampering token ─────────────────────────────────────
   194→   194→
   195→   195→async function generateToken(
   196→   196→  role: string,
   197→   197→  adminFolders: string[],
   198→   198→  salt: string
   199→   199→): Promise<string> {
   200→   200→  const data = `${role}:${adminFolders.sort().join(",")}:${salt}`;
   201→   201→  const buf = await crypto.subtle.digest(
   202→   202→    "SHA-256",
   203→   203→    new TextEncoder().encode(data)
   204→   204→  );
   205→   205→  return Array.from(new Uint8Array(buf))
   206→   206→    .map((b) => b.toString(16).padStart(2, "0"))
   207→   207→    .join("");
   208→   208→}

   210→   210→// ── Google Drive API ─────────────────────────────────────────
   211→   211→
   212→   212→async function driveFetch(
   213→   213→  path: string,
   214→   214→  params: Record<string, string>,
   215→   215→  apiKey: string
   216→   216→): Promise<Record<string, unknown>> {
   217→   217→  const url = new URL(`https://www.googleapis.com/drive/v3${path}`);
   218→   218→  url.searchParams.set("key", apiKey);
   219→   219→  Object.entries(params).forEach(([k, v]) => url.searchParams.set(k, v));
   220→   220→
   221→   221→  const res = await fetch(url.toString());
   222→   222→  if (!res.ok) {
   223→   223→    const body = await res.json().catch(() => ({}));
   224→   224→    throw new Error(
   225→   225→      (body as Record<string, unknown>)?.error
   226→   226→        ? String(((body as Record<string, unknown>).error as Record<string, unknown>)?.message || `Drive API error ${res.status}`)
   227→   227→        : `Drive API error ${res.status}`
   228→   228→    );
   229→   229→  }
   230→   230→  return res.json() as Promise<Record<string, unknown>>;
   231→   231→}

   233→   233→// ── Auth helper ──────────────────────────────────────────────

   235→   235→async function getAuth(
   236→   236→  request: Request,
   237→   237→  env: Env
   238→   238→): Promise<Record<string, unknown> | null> {
   239→   239→  const auth = request.headers.get("Authorization");
   240→   240→  if (!auth?.startsWith("Bearer ")) return null;
   241→   241→  return verifyJWT(auth.slice(7), env.JWT_SECRET);
   242→   242→}

   246→   246→async function handleAdminAuth(
   247→   247→  request: Request,
   248→   248→  env: Env
   249→   249→): Promise<Response> {
   250→   250→  const body = (await request.json()) as { password?: string };
   251→   251→  const { password } = body;
   252→   252→
   253→   253→  if (!password || password !== env.ADMIN_PASSWORD) {
   254→   254→    return json({ error: "Неверный пароль" }, 401);
   255→   255→  }
   256→   256→
   257→   257→  const now = Math.floor(Date.now() / 1000);
   258→   258→  const jwt = await createJWT(
   259→   259→    {
   260→   260→      role: "admin",
   261→   261→      adminFolders: [],
   262→   262→      iat: now,
   263→   263→      exp: now + 7 * 24 * 3600, // 7 дней
   264→   264→    },
   265→   265→    env.JWT_SECRET
   266→   266→  );
   267→   267→
   268→   268→  return json({ token: jwt, role: "admin" });
   269→   269→}

   271→   271→async function handleGuestAuth(
   272→   272→  request: Request,
   273→   273→  env: Env
   274→   274→): Promise<Response> {
   275→   275→  const body = (await request.json()) as {
   276→   276→    keyFile?: string;
   277→   277→    password?: string;
   278→   278→  };
   279→   279→  const { keyFile, password } = body;
   280→   280→
   281→   281→  if (!keyFile || !password) {
   282→   282→    return json({ error: "Не указан ключ-файл или пароль" }, 400);
   283→   283→  }
   284→   284→
   285→   285→  let kf: Record<string, unknown>;
   286→   286→  try {
   287→   287→    kf = (await decrypt(keyFile, password)) as Record<string, unknown>;
   288→   288→  } catch (e) {
   289→   289→    return json(
   290→   290→      { error: e instanceof Error ? e.message : "Ошибка расшифровки" },
   291→   291→      401
   292→   292→    );
   293→   293→  }
   294→   294→
   295→   295→  if (kf.v !== 4) {
   296→   296→    return json({ error: "Неподдерживаемая версия ключ-файла" }, 400);
   297→   297→  }
   298→   298→
   299→   299→  // Проверка анти-тамперинг токена (TOKEN_SALT скрыт на сервере!)
   300→   300→  const expectedToken = await generateToken(
   301→   301→    String(kf.role),
   302→   302→    (kf.adminFolders as string[]) || [],
   303→   303→    env.TOKEN_SALT
   304→   304→  );
   305→   305→  if (kf.token !== expectedToken) {
   306→   306→    return json({ error: "Ключ-файл повреждён или был изменён" }, 401);
   307→   307→  }
   308→   308→
   309→   309→  if (kf.role !== "guest") {
   310→   310→    return json({ error: "Это не гостевой ключ-файл" }, 400);
   311→   311→  }
   312→   312→
   313→   313→  const now = Math.floor(Date.now() / 1000);
   314→   314→  const adminFolders = (kf.adminFolders as string[]) || [];
   315→   315→  const jwt = await createJWT(
   316→   316→    {
   317→   317→      role: "guest",
   318→   318→      adminFolders,
   319→   319→      iat: now,
   320→   320→      exp: now + 30 * 24 * 3600, // 30 дней
   321→   321→    },
   322→   322→    env.JWT_SECRET
   323→   323→  );
   324→   324→
   325→   325→  return json({ token: jwt, role: "guest", adminFolders });
   326→   326→}

   328→   328→async function handleKeyFileGenerate(
   329→   329→  request: Request,
   330→   330→  env: Env
   331→   331→): Promise<Response> {
   332→   332→  const auth = await getAuth(request, env);
   333→   333→  if (!auth || auth.role !== "admin") {
   334→   334→    return json({ error: "Требуется авторизация администратора" }, 403);
   335→   335→  }
   336→   336→
   337→   337→  const body = (await request.json()) as {
   338→   338→    password?: string;
   339→   339→    adminFolders?: string[];
   340→   340→  };
   341→   341→  const { password, adminFolders } = body;
   342→   342→
   343→   343→  if (!password || password.length < 4) {
   344→   344→    return json(
   345→   345→      { error: "Пароль должен быть не менее 4 символов" },
   346→   346→      400
   347→   347→    );
   348→   348→  }
   349→   349→
   350→   350→  // Генерация анти-тамперинг токена (TOKEN_SALT скрыт на сервере!)
   351→   351→  const token = await generateToken(
   352→   352→    "guest",
   353→   353→    adminFolders || [],
   354→   354→    env.TOKEN_SALT
   355→   355→  );
   356→   356→
   357→   357→  const keyFileData = {
   358→   358→    v: 4,
   359→   359→    role: "guest",
   360→   360→    adminFolders: adminFolders || [],
   361→   361→    token,
   362→   362→  };
   363→   363→
   364→   364→  // Шифрование ключ-файла на сервере
   365→   365→  const encrypted = await encrypt(keyFileData, password);
   366→   366→
   367→   367→  return json({ keyFile: encrypted });
   368→   368→}

   370→   370→async function handleFolders(
   371→   371→  request: Request,
   372→   372→  env: Env
   373→   373→): Promise<Response> {
   374→   374→  const auth = await getAuth(request, env);
   375→   375→  if (!auth) {
   376→   376→    return json({ error: "Требуется авторизация" }, 401);
   377→   377→  }
   378→   378→
   379→   379→  const data = await driveFetch(
   380→   380→    "/files",
   381→   381→    {
   382→   382→      q: `'${env.DRIVE_ROOT_FOLDER_ID}' in parents and mimeType = 'application/vnd.google-apps.folder' and trashed = false`,
   383→   383→      fields: "files(id, name, createdTime, parents)",
   384→   384→      pageSize: "200",
   385→   385→      orderBy: "name",
   386→   386→      supportsAllDrives: "true",
   387→   387→      includeItemsFromAllDrives: "true",
   388→   388→    },
   389→   389→    env.DRIVE_API_KEY
   390→   390→  );
   391→   391→
   392→   392→  let folders = (data.files as Record<string, unknown>[]) || [];
   393→   393→
   394→   394→  // Фильтрация папок на сервере! Гость не видит admin-папки.
   395→   395→  if (auth.role === "guest" && (auth.adminFolders as string[])?.length > 0) {
   396→   396→    const adminFolders = auth.adminFolders as string[];
   397→   397→    folders = folders.filter((f) => !adminFolders.includes(f.id as string));
   398→   398→  }
   399→   399→
   400→   400→  return json({ folders });
   401→   401→}

   425→   425→      fields:
   426→   426→        "files(id, name, mimeType, createdTime, modifiedTime, parents, size, webViewLink, webContentLink, thumbnailLink, imageMediaMetadata, videoMediaMetadata)",
   427→   427→      pageSize: "200",
   428→   428→      orderBy: "createdTime desc",
   429→   429→      supportsAllDrives: "true",
   430→   430→      includeItemsFromAllDrives: "true",
   431→   431→    },
   432→   432→    env.DRIVE_API_KEY
   433→   433→  );
   434→   434→
   435→   435→  return json({ files: data.files || [] });
   436→   436→}

   438→   438→async function handleBatchFiles(
   439→   439→  request: Request,
   440→   440→  env: Env
   441→   441→): Promise<Response> {
   442→   442→  const auth = await getAuth(request, env);
   443→   443→  if (!auth) {
   444→   444→    return json({ error: "Требуется авторизация" }, 401);
   445→   445→  }
   446→   446→
   447→   447→  // Parse folder IDs from query params: ?ids=id1,id2,id3
   448→   448→  const url = new URL(request.url);
   449→   449→  const idsParam = url.searchParams.get("ids");
   450→   450→  if (!idsParam) {
   451→   451→    return json({ error: "Не указаны ID папок (параметр ids)" }, 400);
   452→   452→  }
   453→   453→
   454→   454→  const folderIds = idsParam.split(",").filter(Boolean);
   455→   455→  if (folderIds.length === 0) {
   456→   456→    return json({ error: "Пустой список ID папок" }, 400);
   457→   457→  }
   458→   458→
   459→   459→  // Filter out admin-only folders for guest users
   460→   460→  const adminFolders = (auth.adminFolders as string[]) || [];
   461→   461→  const allowedFolderIds = auth.role === "guest" && adminFolders.length > 0
   462→   462→    ? folderIds.filter((id) => !adminFolders.includes(id))
   463→   463→    : folderIds;
   464→   464→
   465→   465→  if (allowedFolderIds.length === 0) {
   466→   466→    return json({ files: {} });
   467→   467→  }
   468→   468→
   469→   469→  // Build a single Drive API query using OR conditions
   470→   470→  // This uses only ONE subrequest instead of N separate requests
   471→   471→  const parentQueries = allowedFolderIds.map((id) => `'${id}' in parents`).join(" or ");
   472→   472→  const query = `(${parentQueries}) and (mimeType contains 'image/' or mimeType contains 'video/') and trashed = false`;
   473→   473→
   474→   474→  const data = await driveFetch(
   475→   475→    "/files",
   476→   476→    {
   477→   477→      q: query,
   478→   478→      fields:
   479→   479→        "files(id, name, mimeType, createdTime, modifiedTime, parents, size, webViewLink, webContentLink, thumbnailLink, imageMediaMetadata, videoMediaMetadata)",
   480→   480→      pageSize: "1000",
   481→   481→      orderBy: "createdTime desc",
   482→   482→      supportsAllDrives: "true",
   483→   483→      includeItemsFromAllDrives: "true",
   484→   484→    },
   485→   485→    env.DRIVE_API_KEY
   486→   486→  );
   487→   487→
   488→   488→  // Group files by their parent folder
   489→   489→  const filesByFolder: Record<string, Record<string, unknown>[]> = {};
   490→   490→  const rawFiles = (data.files as Record<string, unknown>[]) || [];
   491→   491→
   492→   492→  for (const file of rawFiles) {
   493→   493→    const parents = (file.parents as string[]) || [];
   494→   494→    for (const parentId of parents) {
   495→   495→      if (allowedFolderIds.includes(parentId)) {
   496→   496→        if (!filesByFolder[parentId]) filesByFolder[parentId] = [];
   497→   497→        filesByFolder[parentId].push(file);
   498→   498→      }
   499→   499→    }
   500→   500→  }
   501→   501→
   502→   502→  // Ensure all requested folders have an entry (even empty ones)
   503→   503→  for (const folderId of allowedFolderIds) {
   504→   504→    if (!filesByFolder[folderId]) {
   505→   505→      filesByFolder[folderId] = [];
   506→   506→    }
   507→   507→  }
   508→   508→
   509→   509→  return json({ files: filesByFolder });
   510→   510→}

   538→   538→// ── Media proxy — streams files from Google Drive ──────────
   539→   539→
   540→   540→async function handleMedia(
   541→   541→  request: Request,
   542→   542→  env: Env,
   543→   543→  fileId: string
   544→   544→): Promise<Response> {
   545→   545→  // Auth: check token from query param or Authorization header
   546→   546→  let token: string | null = null;
   547→   547→  const url = new URL(request.url);
   548→   548→  if (url.searchParams.has("token")) {
   549→   549→    token = url.searchParams.get("token");
   550→   550→  } else {
   551→   551→    const auth = await getAuth(request, env);
   552→   552→    if (!auth) {
   553→   553→      return json({ error: "Требуется авторизация" }, 401);
   554→   554→    }
   555→   555→    token = request.headers.get("Authorization")?.slice(7) || null;
   556→   556→  }
   557→   557→
   558→   558→  if (!token) {
   559→   559→    return json({ error: "Требуется авторизация" }, 401);
   560→   560→  }
   561→   561→
   562→   562→  // Verify JWT
   563→   563→  const payload = await verifyJWT(token, env.JWT_SECRET);
   564→   564→  if (!payload) {
   565→   565→    return json({ error: "Недействительный токен" }, 401);
   566→   566→  }
   567→   567→
   568→   568→  // ── HEAD request: probe if file is accessible ──
   569→   569→  // The client sends a HEAD request to check if the video URL works
   570→   570→  // before setting it as <video src>. This prevents the native
   571→   571→  // browser error overlay from appearing.
   572→   572→  if (request.method === "HEAD") {
   573→   573→    const probeUrl = `https://www.googleapis.com/drive/v3/files/${fileId}?alt=media&key=${env.DRIVE_API_KEY}&supportsAllDrives=true&includeItemsFromAllDrives=true`;
   574→   574→    try {
   575→   575→      const probeRes = await fetch(probeUrl, {
   576→   576→        headers: { "Range": "bytes=0-1" },
   577→   577→      });
   578→   578→      if (probeRes.ok || probeRes.status === 206) {
   579→   579→        const contentType = probeRes.headers.get("Content-Type") || "application/octet-stream";
   580→   580→        const contentLength = probeRes.headers.get("Content-Range")?.split("/")?.[1] || probeRes.headers.get("Content-Length") || "0";
   581→   581→        return new Response(null, {
   582→   582→          status: 200,
   583→   583→          headers: {
   584→   584→            ...CORS_HEADERS,
   585→   585→            "Content-Type": contentType,
   586→   586→            "Content-Length": contentLength,
   587→   587→            "Accept-Ranges": "bytes",
   588→   588→            "Cache-Control": "no-cache",
   589→   589→            "X-DMGA-Status": "accessible",
   590→   590→          },
   591→   591→        });
   592→   592→      } else {
   593→   593→        let errorMsg = `Drive API error ${probeRes.status}`;
   594→   594→        try {
   595→   595→          const errBody = await probeRes.json();
   596→   596→          errorMsg = (errBody as Record<string, unknown>)?.error
   597→   597→            ? String(((errBody as Record<string, unknown>).error as Record<string, unknown>)?.message || errorMsg)
   598→   598→            : errorMsg;
   599→   599→        } catch {}
   600→   600→        return json({ error: errorMsg }, probeRes.status as 400 | 401 | 403 | 404 | 500);
   601→   601→      }
   602→   602→    } catch (err) {
   603→   603→      return json({ error: `Network error: ${err instanceof Error ? err.message : "unknown"}` }, 502);
   604→   604→    }
   605→   605→  }
   606→   606→
   607→   607→  // ── GET request: stream the file from Google Drive ──
   608→   608→  // Forward Range header for video seeking support
   609→   609→  const driveHeaders: Record<string, string> = {};
   610→   610→  const rangeHeader = request.headers.get("Range");
   611→   611→  if (rangeHeader) {
   612→   612→    driveHeaders["Range"] = rangeHeader;
   613→   613→  }
   614→   614→
   615→   615→  const driveUrl = `https://www.googleapis.com/drive/v3/files/${fileId}?alt=media&key=${env.DRIVE_API_KEY}&supportsAllDrives=true&includeItemsFromAllDrives=true`;
   616→   616→
   617→   617→  const driveRes = await fetch(driveUrl, { headers: driveHeaders });
   618→   618→  if (!driveRes.ok) {
   619→   619→    const body = await driveRes.text().catch(() => "");
   620→   620→    return json(
   621→   621→      { error: `Не удалось получить файл: ${driveRes.status}` },
   622→   622→      driveRes.status as 400 | 401 | 403 | 404 | 500
   623→   623→    );
   624→   624→  }
   625→   625→
   626→   626→  // Stream response with CORS headers
   627→   627→  const contentType = driveRes.headers.get("Content-Type") || "application/octet-stream";
   628→   628→  const contentLength = driveRes.headers.get("Content-Length");
   629→   629→  const contentRange = driveRes.headers.get("Content-Range");
   630→   630→  const acceptRanges = driveRes.headers.get("Accept-Ranges");
   631→   631→  const headers: Record<string, string> = {
   632→   632→    ...CORS_HEADERS,
   633→   633→    "Content-Type": contentType,
   634→   634→    "Cache-Control": "public, max-age=3600", // 1h cache
   635→   635→    "Accept-Ranges": acceptRanges || "bytes",
   636→   636→  };
   637→   637→  if (contentLength) headers["Content-Length"] = contentLength;
   638→   638→  if (contentRange) headers["Content-Range"] = contentRange;
   639→   639→
   640→   640→  // 206 Partial Content for range requests
   641→   641→  const isPartial = driveRes.status === 206;
   642→   642→  return new Response(driveRes.body, {
   643→   643→    status: isPartial ? 206 : 200,
   644→   644→    headers,
   645→   645→  });
   646→   646→}

   648→   648→// ── Thumbnail proxy — returns thumbnail image from Google Drive ──
   649→   649→// Used as fallback when thumbnailLink is not available.
   650→   650→// Supports ?token= query param for auth and ?sz= for size.
   651→   651→
   652→   652→async function handleThumbnail(
   653→   653→  request: Request,
   654→   654→  env: Env,
   655→   655→  fileId: string
   656→   656→): Promise<Response> {
   657→   657→  // Auth: check token from query param or Authorization header
   658→   658→  let token: string | null = null;
   659→   659→  const url = new URL(request.url);
   660→   660→  if (url.searchParams.has("token")) {
   661→   661→    token = url.searchParams.get("token");
   662→   662→  } else {
   663→   663→    const auth = await getAuth(request, env);
   664→   664→    if (!auth) {
   665→   665→      return json({ error: "Требуется авторизация" }, 401);
   666→   666→    }
   667→   667→    token = request.headers.get("Authorization")?.slice(7) || null;
   668→   668→  }
   669→   669→
   670→   670→  if (!token) {
   671→   671→    return json({ error: "Требуется авторизация" }, 401);
   672→   672→  }
   673→   673→
   674→   674→  // Verify JWT
   675→   675→  const payload = await verifyJWT(token, env.JWT_SECRET);
   676→   676→  if (!payload) {
   677→   677→    return json({ error: "Недействительный токен" }, 401);
   678→   678→  }
   679→   679→
   680→   680→  // Get thumbnail size from query param (default w400)
   681→   681→  const sizeParam = url.searchParams.get("sz") || "w400";
   682→   682→  const size = parseInt(sizeParam.replace("w", ""), 10) || 400;
   683→   683→
   684→   684→  // First, try to get the thumbnailLink from the Drive API
   685→   685→  try {
   686→   686→    const fileData = await driveFetch(
   687→   687→      `/files/${fileId}`,
   688→   688→      {
   689→   689→        fields: "thumbnailLink,mimeType",
   690→   690→        supportsAllDrives: "true",
   691→   691→        includeItemsFromAllDrives: "true",
   692→   692→      },
   693→   693→      env.DRIVE_API_KEY
   694→   694→    );
   695→   695→
   696→   696→    const thumbnailLink = fileData.thumbnailLink as string | undefined;
   697→   697→    if (thumbnailLink) {
   698→   698→      // Replace the size suffix in the thumbnail URL
   699→   699→      const sizedUrl = thumbnailLink.replace(/=s\d+$/, `=s${size}`);
   700→   700→      const thumbRes = await fetch(sizedUrl);
   701→   701→      if (thumbRes.ok) {
   702→   702→        const contentType = thumbRes.headers.get("Content-Type") || "image/jpeg";
   703→   703→        const body = await thumbRes.arrayBuffer();
   704→   704→        return new Response(body, {
   705→   705→          status: 200,
   706→   706→          headers: {
   707→   707→            ...CORS_HEADERS,
   708→   708→            "Content-Type": contentType,
   709→   709→            "Cache-Control": "public, max-age=3600",
   710→   710→            "Access-Control-Allow-Origin": "*",
   711→   711→          },
   712→   712→        });
   713→   713→      }
   714→   714→    }
   715→   715→  } catch {
   716→   716→    // thumbnailLink approach failed, try direct thumbnail
   717→   717→  }
   718→   718→
   719→   719→  // Fallback: use drive.google.com/thumbnail
   720→   720→  const thumbUrl = `https://drive.google.com/thumbnail?id=${fileId}&sz=w${size}`;
   721→   721→  try {
   722→   722→    const thumbRes = await fetch(thumbUrl, {
   723→   723→      headers: { "Authorization": `Bearer ${env.DRIVE_API_KEY}` },
   724→   724→    });
   725→   725→    if (thumbRes.ok) {
   726→   726→      const contentType = thumbRes.headers.get("Content-Type") || "image/jpeg";
   727→   727→      const body = await thumbRes.arrayBuffer();
   728→   728→      return new Response(body, {
   729→   729→        status: 200,
   730→   730→        headers: {
   731→   731→          ...CORS_HEADERS,
   732→   732→          "Content-Type": contentType,
   733→   733→          "Cache-Control": "public, max-age=3600",
   734→   734→        },
   735→   735→      });
   736→   736→    }
   737→   737→  } catch {
   738→   738→    // All thumbnail methods failed
   739→   739→  }
   740→   740→
   741→   741→  return json({ error: "Не удалось получить миниатюру" }, 404);
   742→   742→}

   745→   745→
   746→   746→export default {
   747→   747→  async fetch(request: Request, env: Env): Promise<Response> {
   748→   748→    // CORS preflight
   749→   749→    if (request.method === "OPTIONS") {
   750→   750→      return new Response(null, { headers: CORS_HEADERS });
   751→   751→    }
   752→   752→
   753→   753→    const url = new URL(request.url);
   754→   754→
   755→   755→    try {
   756→   756→      // Авторизация
   757→   757→      if (url.pathname === "/auth/admin" && request.method === "POST") {
   758→   758→        return handleAdminAuth(request, env);
   759→   759→      }
   760→   760→      if (url.pathname === "/auth/guest" && request.method === "POST") {
   761→   761→        return handleGuestAuth(request, env);
   762→   762→      }
   763→   763→
   764→   764→      // Генерация ключ-файлов (только для админа)
   765→   765→      if (url.pathname === "/keyfile/generate" && request.method === "POST") {
   766→   766→        return handleKeyFileGenerate(request, env);
   767→   767→      }
   768→   768→
   769→   769→      // Данные — требуют авторизацию
   770→   770→      if (url.pathname === "/folders" && request.method === "GET") {
   771→   771→        return handleFolders(request, env);
   772→   772→      }
   773→   773→      if (url.pathname === "/files/root" && request.method === "GET") {
   774→   774→        return handleRootFiles(request, env);
   775→   775→      }
   776→   776→      if (url.pathname === "/files/batch" && request.method === "GET") {
   777→   777→        return handleBatchFiles(request, env);
   778→   778→      }
   779→   779→      if (url.pathname.startsWith("/files/") && request.method === "GET") {
   780→   780→        const folderId = url.pathname.slice(7);
   781→   781→        return handleFiles(request, env, folderId);
   782→   782→      }
   783→   783→
   784→   784→      // Миниатюры — проксирует thumbnails из Google Drive
   785→   785→      if (url.pathname.startsWith("/thumbnail/") && request.method === "GET") {
   786→   786→        const fileId = url.pathname.slice(11);
   787→   787→        return handleThumbnail(request, env, fileId);
   788→   788→      }
   789→   789→
   790→   790→      // Медиа-прокси — стримит файлы Google Drive через Worker
   791→   791→      // Используется для видео и полноразмерных изображений
   792→   792→      // Поддерживает GET (стриминг) и HEAD (проверка доступности)
   793→   793→      if (url.pathname.startsWith("/media/") && (request.method === "GET" || request.method === "HEAD")) {
   794→   794→        const fileId = url.pathname.slice(7);
   795→   795→        return handleMedia(request, env, fileId);
   796→   796→      }
   797→   797→
   798→   798→      // Health check
   799→   799→      if (url.pathname === "/") {
   800→   800→        return json({ status: "ok", service: "DMGA API Proxy" });
   801→   801→      }
   802→   802→
   803→   803→      return json({ error: "Not found" }, 404);
   804→   804→    } catch (err: unknown) {
   805→   805→      const message = err instanceof Error ? err.message : "Internal server error";
   806→   806→      return json({ error: message }, 500);
   807→   807→    }
   808→   808→  },
   809→   809→};
